cmake_minimum_required(VERSION 3.16)
project(picoa26_host_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
option(A26_SANITIZE "Run address/undefined-behaviour sanitizers" ON)
set(CORE ${CMAKE_CURRENT_LIST_DIR}/../core2600)
add_library(a26core STATIC ${CORE}/cpu.c ${CORE}/video.c ${CORE}/memory.c
    ${CORE}/timer.c ${CORE}/input.c ${CORE}/delayed_writer.c
    ${CORE}/cartridge.c ${CORE}/a26_core.c ../atari/atari_audio.c)
target_compile_definitions(a26core PUBLIC A26_HOST_TEST=1)
target_include_directories(a26core PUBLIC ${CORE} ..)
add_executable(core_tests core_tests.c)
target_link_libraries(core_tests PRIVATE a26core)
add_executable(cpu_tests cpu_tests.c ${CORE}/cpu.c)
add_executable(i2s_tests i2s_tests.c)
target_include_directories(i2s_tests PRIVATE ..)
add_executable(blitter_tests blitter_tests.c)
target_include_directories(blitter_tests PRIVATE .. stubs)
target_compile_definitions(cpu_tests PRIVATE A26_HOST_TEST=1)
target_include_directories(cpu_tests PRIVATE ${CORE})
foreach(t a26core core_tests cpu_tests i2s_tests blitter_tests)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wno-unused-variable -Wno-unused-parameter
        -Wno-sign-compare -fno-strict-aliasing -g)
    if(A26_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${t} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
add_test(NAME core COMMAND core_tests)
add_test(NAME cpu COMMAND cpu_tests)
add_test(NAME i2s COMMAND i2s_tests)
add_test(NAME blitter COMMAND blitter_tests)
add_library(cpu_vectors SHARED cpu_harness.c ${CORE}/cpu.c)
target_compile_definitions(cpu_vectors PRIVATE A26_HOST_TEST=1)
target_include_directories(cpu_vectors PRIVATE ${CORE})
